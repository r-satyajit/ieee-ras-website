Rover point cloud alignment has been integrated with the new ROS2 Humble nav2 costmap layer. The loop closure drift is down to < 1.2cm. Take a look at the revised SLAM lifecycle node:
ros2_autonomous_nav_node.py 
import rclpy
from rclpy.node import Node
from sensor_msgs.msg import LaserScan, PointCloud2
from geometry_msgs.msg import Twist

class AutonomousRoverNavigator(Node):
    def __init__(self):
        super().__init__('rover_slam_navigator')
        self.cmd_pub = self.create_publisher(Twist, '/cmd_vel', 10)
        self.scan_sub = self.create_subscription(LaserScan, '/scan', self.lidar_callback, 10)
        self.get_logger().info('[IEEE_RAS] Lidar-SLAM Navigation Node Initialized')

    def lidar_callback(self, msg):
        # Compute obstacle gradient and kinematic trajectory
        min_dist = min(msg.ranges)
        twist = Twist()
        twist.linear.x = 0.45 if min_dist > 0.8 else 0.0
        twist.angular.z = 0.0 if min_dist > 0.8 else 0.35
        self.cmd_pub.publish(twist)
